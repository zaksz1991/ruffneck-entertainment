/**
 * RuffNeck Entertainment — Automated Email Delivery
 * ==================================================
 * Handles TWO types of automated emails from your website:
 *   1. Purchase delivery — sent when someone buys a product on store.html
 *   2. Lead magnet delivery — sent when someone requests the free "20 AI Prompts" guide on index.html
 *
 * 100% FREE — runs entirely on Google's infrastructure using your Gmail account.
 *
 * ── HOW TO UPDATE YOUR EXISTING DEPLOYMENT ──
 * 1. Go to script.google.com and open your existing "RuffNeck PDF Delivery" project
 * 2. Select ALL the old code and delete it
 * 3. Paste this ENTIRE file in its place
 * 4. Click the disk icon to save
 * 5. Click "Deploy" then "Manage deployments"
 * 6. Click the pencil/edit icon next to your existing active deployment
 * 7. Under "Version", select "New version"
 * 8. Click "Deploy"
 *
 * IMPORTANT: You do NOT need a new URL — updating the existing deployment
 * keeps the same Web App URL, so nothing needs to change in your website files.
 */

function doPost(e) {
  try {
    const data = JSON.parse(e.postData.contents);
    const requestType = data.type || 'purchase';

    if (requestType === 'lead_magnet') {
      return handleLeadMagnet(data);
    }
    return handlePurchase(data);

  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

// ======================================================
// LEAD LOG - keeps a simple running list of every lead/customer in a Google Sheet
// so you can see your growing email list. Auto-creates the sheet on first use.
// ======================================================
function logToSheet(type, name, email, detail) {
  try {
    const props = PropertiesService.getScriptProperties();
    let sheetId = props.getProperty('RUFFNECK_LEADS_SHEET_ID');
    let sheet;

    if (!sheetId) {
      const ss = SpreadsheetApp.create('RuffNeck Entertainment - Leads & Customers');
      sheetId = ss.getId();
      props.setProperty('RUFFNECK_LEADS_SHEET_ID', sheetId);
      sheet = ss.getSheets()[0];
      sheet.appendRow(['Date', 'Type', 'Name', 'Email', 'Detail']);
      sheet.setFrozenRows(1);
    } else {
      sheet = SpreadsheetApp.openById(sheetId).getSheets()[0];
    }

    sheet.appendRow([new Date(), type, name, email, detail]);
  } catch (err) {
    console.error('Sheet logging failed: ' + err.toString());
  }
}
